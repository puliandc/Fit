# Fit iOS 应用调试指南

## 概述

本指南详细介绍 Fit iOS 应用的调试技术、工具使用和常见问题解决方法，帮助开发者高效地进行应用调试和问题排查。

## 调试工具概览

### 1. Xcode 内置调试工具

#### 断点调试
- **断点设置**: 在代码行号左侧点击设置断点
- **条件断点**: 右键断点设置触发条件
- **异常断点**: 捕获程序异常
- **符号断点**: 根据方法名设置断点

#### 调试导航器
- **变量查看**: 查看当前作用域内的变量值
- **调用栈**: 查看函数调用层级
- **线程查看**: 查看多线程执行状态
- **内存查看**: 检查内存使用情况

#### 控制台输出
```swift
// 基本输出
print("调试信息")

// 带级别的日志输出

// 调试专用输出（仅在 Debug 模式下显示）
#if DEBUG
print("🐛 DEBUG: 调试信息")
#endif
```

### 2. 外部调试工具

#### Instruments 工具套件
- **Time Profiler**: CPU 性能分析
- **Leaks**: 内存泄漏检测
- **Allocations**: 内存分配跟踪
- **Energy Log**: 电池消耗分析
- **Network**: 网络性能分析

#### 第三方工具
- **Charles Proxy**: 网络请求调试
- **Reveal**: UI 层级调试
- **Crashlytics**: 崩溃报告收集

## 日志系统

### 1. 日志级别定义

Fit 应用使用统一的日志格式：

```swift
// 日志级别和格式
extension String {
    static func log(_ message: String, level: LogLevel = .info, file: String = #file, function: String = #function, line: Int = #line) {
        let fileName = (file as NSString).lastPathComponent
        let timestamp = DateFormatter.logFormatter.string(from: Date())

        let logMessage: String
        switch level {
        case .debug:
            logMessage = "🐛 DEBUG: [\(fileName):\(line)] \(function) - \(message)"
        case .info:
            logMessage = "ℹ️ INFO: [\(fileName):\(line)] \(function) - \(message)"
        case .warning:
            logMessage = "⚠️ WARNING: [\(fileName):\(line)] \(function) - \(message)"
        case .error:
            logMessage = "🚨 ERROR: [\(fileName):\(line)] \(function) - \(message)"
        }

        print(logMessage)
    }
}

enum LogLevel {
    case debug, info, warning, error
}

extension DateFormatter {
    static let logFormatter: DateFormatter = {
        let formatter = DateFormatter()
        formatter.dateFormat = "HH:mm:ss.SSS"
        return formatter
    }()
}
```

### 2. 使用示例

```swift
class WorkoutViewModel: ObservableObject {
    func startExercise() {
        .log("开始练习: \(currentExercise.name)", level: .info)

        guard currentExerciseIndex < workoutPlan.exercises.count else {
            .log("练习索引超出范围", level: .error)
            return
        }

        isExerciseActive = true
        .log("练习状态已更新: \(isExerciseActive)", level: .debug)
    }

    func completeExerciseWith(actualReps: Int, actualWeight: Double) {
        .log("完成练习参数 - 次数: \(actualReps), 重量: \(actualWeight)", level: .info)

        // 验证参数
        if actualReps <= 0 {
            .log("无效的次数: \(actualReps)", level: .warning)
            return
        }

        // 完成逻辑
        let completedSet = CompletedSet(
            exerciseSetId: currentExerciseSet.id,
            actualReps: actualReps,
            actualWeight: actualWeight,
            completedAt: Date()
        )
        completedSets.append(completedSet)

        .log("已记录完成组数: \(completedSets.count)", level: .debug)
    }
}
```

## 常见问题调试

### 1. 数据绑定问题

#### 问题：UI 不更新
```swift
// 确保使用 @Published 属性
class WorkoutViewModel: ObservableObject {
    @Published var currentExerciseIndex: Int = 0  // ✅ 正确
    // var currentExerciseIndex: Int = 0        // ❌ 错误

    func updateProgress() {
        // 确保在主线程更新
        DispatchQueue.main.async {
            self.currentExerciseIndex += 1
        }
    }
}
```

#### 解决方案：
```swift
// 在 View 中正确绑定
struct WorkoutScreen: View {
    @EnvironmentObject var workoutViewModel: WorkoutViewModel

    var body: some View {
        VStack {
            Text("当前练习: \(workoutViewModel.currentExercise.name)")
            Text("进度: \(Int(workoutViewModel.progress * 100))%")
        }
        .onReceive(workoutViewModel.$currentExerciseIndex) { index in
            print("UI 更新: currentExerciseIndex = \(index)")
        }
    }
}
```

### 2. 计时器问题

#### 问题：计时器不准确或不响应
```swift
class WorkoutViewModel: ObservableObject {
    private var exerciseTimer: Timer?
    private var restTimer: Timer?

    func startExerciseTimer() {
        // 清理之前的计时器
        exerciseTimer?.invalidate()

        exerciseTimer = Timer.scheduledTimer(withTimeInterval: 1.0, repeats: true) { [weak self] _ in
            DispatchQueue.main.async {
                self?.exerciseElapsedTime += 1
                print("⏱️ 计时器更新: \(self?.exerciseElapsedTime ?? 0)秒")
            }
        }

        // 确保计时器在主运行循环中运行
        RunLoop.main.add(exerciseTimer!, forMode: .common)
    }

    func pauseExercise() {
        exerciseTimer?.invalidate()
        exerciseTimer = nil
        print("⏸️ 计时器已暂停")
    }

    deinit {
        // 确保在对象销毁时清理计时器
        exerciseTimer?.invalidate()
        restTimer?.invalidate()
        print("🗑️ WorkoutViewModel 销毁，计时器已清理")
    }
}
```

### 3. 文件处理问题

#### 问题：JSON 解析失败
```swift
func handleJSONParsingError(_ error: Error) {
    print("🚨 JSON 解析错误: \(error.localizedDescription)")

    if let parsingError = error as? DecodingError {
        switch parsingError {
        case .dataCorrupted(let context):
            print("📄 数据损坏: \(context.debugDescription)")
        case .keyNotFound(let key, let context):
            print("🔑 键缺失: \(key.stringValue) in \(context.codingPath)")
        case .typeMismatch(let type, let context):
            print("🔄 类型不匹配: \(type) in \(context.codingPath)")
        case .valueNotFound(let type, let context):
            print("❓ 值缺失: \(type) in \(context.codingPath)")
        @unknown default:
            print("❓ 未知解析错误")
        }
    }
}
```

#### 解决方案：
```swift
func parseWorkoutPlan(from data: Data) -> WorkoutPlan? {
    do {
        let decoder = JSONDecoder()
        decoder.dateDecodingStrategy = .iso8601
        let workoutPlan = try decoder.decode(WorkoutPlan.self, from: data)

        print("✅ JSON 解析成功: \(workoutPlan.name)")
        return workoutPlan

    } catch {
        handleJSONParsingError(error)
        return nil
    }
}
```

## 内存调试

### 1. 内存泄漏检测

```swift
// 使用 Instruments Leaks 工具
// 或者在代码中添加内存监控
class MemoryMonitor {
    static func logMemoryUsage() {
        let info = mach_task_basic_info()
        var count = mach_msg_type_number_t(MemoryLayout<mach_task_basic_info>.size)/4

        let kerr: kern_return_t = withUnsafeMutablePointer(to: &info) {
            $0.withMemoryRebound(to: integer_t.self, capacity: 1) {
                task_info(mach_task_self_,
                         task_flavor_t(MACH_TASK_BASIC_INFO),
                         $0,
                         &count)
            }
        }

        if kerr == KERN_SUCCESS {
            let usedMB = Double(info.resident_size) / 1024.0 / 1024.0
            print("💾 内存使用: \(String(format: "%.2f", usedMB)) MB")
        }
    }
}
```

### 1. 网络请求监控

```swift
class NetworkLogger {
    static func logRequest(_ request: URLRequest) {
        print("🌐 发送请求:")
        print("  URL: \(request.url?.absoluteString ?? "unknown")")
        print("  Method: \(request.httpMethod ?? "unknown")")
        print("  Headers: \(request.allHTTPHeaderFields ?? [:])")

        if let body = request.httpBody,
           let bodyString = String(data: body, encoding: .utf8) {
            print("  Body: \(bodyString)")
        }
    }

    static func logResponse(_ response: URLResponse, data: Data?) {
        print("📡 收到响应:")
        print("  URL: \(response.url?.absoluteString ?? "unknown")")

        if let httpResponse = response as? HTTPURLResponse {
            print("  Status: \(httpResponse.statusCode)")
            print("  Headers: \(httpResponse.allHeaderFields)")
        }

        if let data = data,
           let responseString = String(data: data, encoding: .utf8) {
            print("  Body: \(responseString.prefix(500))...") // 只显示前500字符
        }
    }
}
```

// 使用统一的日志管理器
class Logger {
    enum Level: String, CaseIterable {
        case debug = "🐛 DEBUG"
        case info = "ℹ️ INFO"
        case warning = "⚠️ WARNING"
        case error = "🚨 ERROR"
    }

    static func log(_ message: String, level: Level = .info, file: String = #file, line: Int = #line) {
        #if DEBUG
        let fileName = (file as NSString).lastPathComponent
        print("\(level.rawValue) [\(fileName):\(line)] \(message)")
        #endif
    }
}
