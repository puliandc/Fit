### Q3: @Published 属性更新但 UI 不刷新

**问题描述**: ViewModel 中的 @Published 属性更新了，但对应的 UI 没有刷新。

**可能原因**:
- 未在主线程更新 UI
- 视图绑定错误
- 对象引用问题

**解决方案**:
```swift
// 1. 确保在主线程更新 UI
class WorkoutViewModel: ObservableObject {
    @Published var currentExerciseIndex: Int = 0

    func updateProgress() {
        // 确保在主线程更新
        DispatchQueue.main.async {
            self.currentExerciseIndex += 1
        }
    }
}

// 2. 检查视图绑定
struct WorkoutScreen: View {

        VStack {

// 3. 使用 objectWillChange 手动触发更新
class WorkoutViewModel: ObservableObject {
```

### Q4: 计时器不准确或不响应

**问题描述**: 计时器显示的时间不准确，或者停止后不响应。

**可能原因**:
- 计时器未在主运行循环中运行
- 内存泄漏导致计时器失效
- 应用进入后台影响计时器

**解决方案**:
```swift
class WorkoutViewModel: ObservableObject {
    private var exerciseTimer: Timer?
    private var restTimer: Timer?

    func startExerciseTimer() {
        // 清理之前的计时器
        exerciseTimer?.invalidate()

        // 在主线程创建计时器
        DispatchQueue.main.async {
            self.exerciseTimer = Timer.scheduledTimer(withTimeInterval: 1.0, repeats: true) { [weak self] _ in
                guard let self = self else { return }

                DispatchQueue.main.async {
                    self.exerciseElapsedTime += 1
                }
            }

            // 确保计时器在主运行循环中运行
            if let timer = self.exerciseTimer {
                RunLoop.main.add(timer, forMode: .common)
            }
        }
    }

    func pauseExercise() {
        exerciseTimer?.invalidate()
        exerciseTimer = nil
        print("计时器已暂停")
    }

    deinit {
        // 确保在对象销毁时清理计时器
        exerciseTimer?.invalidate()
        restTimer?.invalidate()
        print("ViewModel 销毁，计时器已清理")
    }
}
```

### Q5: JSON 解析失败

**问题描述**: 从文件读取的 JSON 数据解析失败。

**可能原因**:
- JSON 格式不正确
- 数据模型与 JSON 结构不匹配
- 文件编码问题

**解决方案**:
```swift
func parseWorkoutPlan(from data: Data) -> WorkoutPlan? {
    do {
        let decoder = JSONDecoder()
        decoder.dateDecodingStrategy = .iso8601
        decoder.keyDecodingStrategy = .convertFromSnakeCase

        let workoutPlan = try decoder.decode(WorkoutPlan.self, from: data)
        print("✅ JSON 解析成功: \(workoutPlan.name)")
        return workoutPlan

    } catch let DecodingError.dataCorrupted(context) {
        print("🚨 JSON 数据损坏: \(context.debugDescription)")
        print("🚨 路径: \(context.codingPath)")
    } catch let DecodingError.keyNotFound(key, context) {
        print("🚨 缺少键: \(key.stringValue)")
        print("🚨 路径: \(context.codingPath)")
    } catch let DecodingError.typeMismatch(type, context) {
        print("🚨 类型不匹配: \(type)")
        print("🚨 路径: \(context.codingPath)")
    } catch {
        print("🚨 解析失败: \(error.localizedDescription)")
    }

    return nil
}

// 调试 JSON 数据
func debugJSONData(_ data: Data) {
    if let jsonString = String(data: data, encoding: .utf8) {
        print("📄 JSON 数据预览:")
        print(jsonString.prefix(1000)) // 只显示前1000字符
    }
}
```

```swift
// 1. 检查文件权限
func checkFilePermissions(at url: URL) -> Bool {
    guard url.startAccessingSecurityScopedResource() else {
        print("🚨 无法获取文件访问权限")
        return false
    }
    defer { url.stopAccessingSecurityScopedResource() }
    return true
}

// 2. 验证文件格式
func validateFileFormat(at url: URL) -> Bool {
    guard url.pathExtension.lowercased() == "json" else {
        print("🚨 文件格式不正确，需要 JSON 文件")
        return false
    }
    return true
}

// 3. 添加详细的错误处理
func loadWorkoutPlan(from url: URL) async {
    print("📱 开始读取文件: \(url.lastPathComponent)")

    // 检查文件权限
    guard checkFilePermissions(at: url) else {
        DispatchQueue.main.async {
            self.errorMessage = "无法访问文件，请检查文件权限"
        }
        return
    }

    // 验证文件格式
    guard validateFileFormat(at: url) else {
        DispatchQueue.main.async {
            self.errorMessage = "文件格式不正确，请选择 JSON 文件"
        }
        return
    }

    do {
        let data = try Data(contentsOf: file)
        print("📊 文件大小: \(data.count) 字节")

        if let workoutPlan = parseWorkoutPlan(from: data) {
            DispatchQueue.main.async {
                self.currentWorkoutPlan = workoutPlan
                print("✅ 训练计划加载成功")
            }
        } else {
            DispatchQueue.main.async {
                self.errorMessage = "无法解析训练计划，请检查文件格式"
            }
        }
    } catch {
        DispatchQueue.main.async {
            self.errorMessage = "读取文件失败: \(error.localizedDescription)"
        }
    }
}
```

### Q7: 训练进度显示不正确

**问题描述**: 训练进度条显示的百分比与实际完成情况不符。

**可能原因**:
- 进度计算逻辑错误
- 组数统计问题
- 数据同步延迟

**解决方案**:
```swift
class WorkoutViewModel: ObservableObject {
    var progress: Double {
        // 确保计算逻辑正确
        let totalSets = workoutPlan.exercises.count
        guard totalSets > 0 else { return 0.0 }

        let completedSetsCount = completedSets.count
        let progressValue = Double(completedSetsCount) / Double(totalSets)

        // 防止进度超过100%
        let clampedProgress = min(max(progressValue, 0.0), 1.0)

        print("📊 进度计算: \(completedSetsCount)/\(totalSets) = \(clampedProgress)")
        return clampedProgress
    }

    func completeExerciseWith(actualReps: Int, actualWeight: Double) {
        print("🏋️ 完成练习: \(currentExercise.name)")

        // 防止重复完成
        let completedSetsCount = completedSets.count
        let totalSets = workoutPlan.exercises.count

        if completedSetsCount >= totalSets {
            print("⚠️ 训练已完成，忽略额外的完成操作")
            return
        }

        // 记录完成的组
        let completedSet = CompletedSet(
            exerciseSetId: currentExerciseSet.id,
            actualReps: actualReps,
            actualWeight: actualWeight,
            completedAt: Date()
        )
        completedSets.append(completedSet)

        // 手动触发 UI 更新
        objectWillChange.send()

        print("✅ 已记录完成组数: \(completedSets.count)")
        print("📊 当前进度: \(Int(progress * 100))%")
    }
}
```

### Q8: 应用进入后台后计时器停止

**问题描述**: 当应用进入后台时，计时器停止工作。

**可能原因**:
- iOS 系统限制后台执行
- 计时器未正确处理应用状态变化

**解决方案**:
```swift
class WorkoutViewModel: ObservableObject {
    private var backgroundTask: UIBackgroundTaskIdentifier = .invalid

    // 监听应用状态变化
    func setupAppStateObservers() {
        NotificationCenter.default.addObserver(
            forName: UIApplication.didEnterBackgroundNotification,
            object: nil,
            queue: .main
        ) { [weak self] _ in
            self?.appDidEnterBackground()
        }

        NotificationCenter.default.addObserver(
            forName: UIApplication.willEnterForegroundNotification,
            object: nil,
            queue: .main
        ) { [weak self] _ in
            self?.appWillEnterForeground()
        }
    }

    func appDidEnterBackground() {
        // 开始后台任务
        backgroundTask = UIApplication.shared.beginBackgroundTask {
            self.endBackgroundTask()
        }

        print("📱 应用进入后台，开始后台任务")
    }

    func appWillEnterForeground() {
        // 结束后台任务
        endBackgroundTask()

        // 重置计时器
        resetTimerIfNeeded()

        print("📱 应用回到前台")
    }

    private func endBackgroundTask() {
        if backgroundTask != .invalid {
            UIApplication.shared.endBackgroundTask(backgroundTask)
            backgroundTask = .invalid
        }
    }

    func resetTimerIfNeeded() {
        guard let startTime = workoutStartTime else { return }

        // 计算应该经过的时间
        let elapsedTime = Int(Date().timeIntervalSince(startTime))

        if isExerciseActive {
            exerciseElapsedTime = elapsedTime
        }

        print("⏰ 计时器已重置: \(elapsedTime)秒")
    }
}
```

### Q9: 内存使用过高

**问题描述**: 应用使用过程中内存占用持续增长。

**可能原因**:
- 循环引用
- 计时器未正确清理
- 大量数据缓存

**解决方案**:
```swift
// 1. 使用 weak 引用避免循环引用
class WorkoutViewModel: ObservableObject {
    weak var delegate: WorkoutViewModelDelegate?

    private var timer: Timer?

    func startTimer() {
        timer = Timer.scheduledTimer(withTimeInterval: 1.0, repeats: true) { [weak self] _ in
            self?.updateTimer()
        }
    }

    deinit {
        timer?.invalidate()
        print("🗑️ WorkoutViewModel 销毁")
    }
}

// 2. 及时清理资源
class NavigationManager: ObservableObject {
    func startWorkout(_ plan: WorkoutPlan) {
        // 清理之前的 ViewModel
        currentWorkoutViewModel = nil

        // 创建新的 ViewModel
        currentWorkoutViewModel = WorkoutViewModel(workoutPlan: plan)
    }

    func endWorkout() {
        // 清理 ViewModel
        currentWorkoutViewModel = nil
    }
}

// 3. 监控内存使用
func logMemoryUsage() {
    let info = mach_task_basic_info()
    var count = mach_msg_type_number_t(MemoryLayout<mach_task_basic_info>.size)/4

    let kerr: kern_return_t = withUnsafeMutablePointer(to: &info) {
        $0.withMemoryRebound(to: integer_t.self, capacity: 1) {
            task_info(mach_task_self_,
                     task_flavor_t(MACH_TASK_BASIC_INFO),
                     $0,
                     &count)
        }
    }

    if kerr == KERN_SUCCESS {
        let usedMB = Double(info.resident_size) / 1024.0 / 1024.0
        print("💾 内存使用: \(String(format: "%.2f", usedMB)) MB")
    }
}
```

### Q10: 文件选择器不显示

**问题描述**: 文件选择器不显示或无法正常工作。

**可能原因**:
- Info.plist 缺少权限配置
- 文件访问权限问题
- iOS 版本兼容性

**解决方案**:
```swift
// 1. 检查 Info.plist 配置
// 确保添加了以下权限：
// - NSDocumentsFolderUsageDescription
// - LSSupportsOpeningDocumentsInPlace

// 2. 正确使用文件选择器
struct FilePickerView: UIViewControllerRepresentable {
    @Binding var isPresented: Bool
    let onFileSelected: (URL) -> Void
    let onError: (String) -> Void

    func makeUIViewController(context: Context) -> UIDocumentPickerViewController {
        let picker = UIDocumentPickerViewController(
            forOpeningContentTypes: [.json],
            asCopy: false
        )
        picker.delegate = context.coordinator
        picker.allowsMultipleSelection = false
        return picker
    }

    func updateUIViewController(_ uiViewController: UIDocumentPickerViewController, context: Context) {}

    func makeCoordinator() -> Coordinator {
        Coordinator(self)
    }

    class Coordinator: NSObject, UIDocumentPickerDelegate {
        let parent: FilePickerView

        init(_ parent: FilePickerView) {
            self.parent = parent
        }

        func documentPicker(_ controller: UIDocumentPickerViewController, didPickDocumentsAt urls: [URL]) {
            guard let url = urls.first else {
                parent.onError("未选择文件")
                return
            }

            print("📁 选择文件: \(url.lastPathComponent)")
            parent.onFileSelected(url)
            parent.isPresented = false
        }

        func documentPickerWasCancelled(_ controller: UIDocumentPickerViewController) {
            print("❌ 用户取消文件选择")
            parent.isPresented = false
        }
    }
}

// 3. 在 View 中正确使用
struct MainScreen: View {
    @State private var showFilePicker = false

    var body: some View {
        VStack {
            Button("选择文件") {
                showFilePicker = true
            }
        }
        .sheet(isPresented: $showFilePicker) {
            FilePickerView(
                isPresented: $showFilePicker,
                onFileSelected: { url in
                    handleFileSelection(url)
                },
                onError: { error in
                    showErrorMessage(error)
                }
            )
        }
    }
}
```
