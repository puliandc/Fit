// src/utils/performance.ts
export const PerformanceOptimizer = {
  // 图片懒加载优化
  setupLazyLoading() {
    if ('IntersectionObserver' in window) {
      const imageObserver = new IntersectionObserver((entries) => {
        entries.forEach(entry => {
          if (entry.isIntersecting) {
            const img = entry.target as HTMLImageElement;

            // 渐进式加载
            const tempImg = new Image();
            tempImg.onload = () => {
              img.src = tempImg.src;
              img.classList.add('loaded');
              imageObserver.unobserve(img);
            };
            tempImg.src = img.dataset.src!;

            // 低质量占位图
            if (img.dataset.lowsrc) {
              img.src = img.dataset.lowsrc;
            }
          }
        });
      }, {
        rootMargin: '50px 0px',
        threshold: 0.1
      });

      document.querySelectorAll('img[data-src]').forEach(img => {
        imageObserver.observe(img);
      });
    }
  },

  // 动画性能优化
  optimizeAnimations() {
    // 使用will-change优化动画性能
    const animatedElements = document.querySelectorAll('[data-animate]');
    animatedElements.forEach(el => {
      const element = el as HTMLElement;
      element.style.willChange = 'transform, opacity';

      // 动画完成后清理will-change
      element.addEventListener('animationend', () => {
        element.style.willChange = 'auto';
      });
    });

    // 减少重绘和重排
    const styleElement = document.createElement('style');
    styleElement.textContent = `
      .glass-card {
        backface-visibility: hidden;
        perspective: 1000px;
        transform: translateZ(0);
      }

      @keyframes fade-in {
        from { opacity: 0; transform: translateY(20px); }
        to { opacity: 1; transform: translateY(0); }
      }

      .animate-optimized {
        animation: fade-in 0.3s cubic-bezier(0.4, 0, 0.2, 1) forwards;
      }
    `;
    document.head.appendChild(styleElement);
  },

  // 内存管理
  setupMemoryManagement() {
    // 定期清理未使用的资源
    const cleanup = () => {
      // 清理音频缓存
      if (window.gc) window.gc();

      // 清理事件监听器
      const elements = document.querySelectorAll('[data-auto-cleanup]');
      elements.forEach(el => {
        const clone = el.cloneNode(true);
        el.parentNode?.replaceChild(clone, el);
      });
    };

    // 每5分钟清理一次
    setInterval(cleanup, 5 * 60 * 1000);

    // 页面隐藏时清理
    document.addEventListener('visibilitychange', () => {
      if (document.hidden) {
        cleanup();
      }
    });
  },

  // 网络优化
  setupNetworkOptimization() {
    // Service Worker注册
    if ('serviceWorker' in navigator) {
      navigator.serviceWorker.register('/sw.js')
        .then(registration => {
          console.log('Service Worker registered:', registration);
        })
        .catch(error => {
          console.warn('Service Worker registration failed:', error);
        });
    }

    // 资源预加载
    const criticalResources = [
      '/audio/start-exercise.mp3',
      '/audio/rest-time.mp3',
      '/audio/workout-complete.mp3'
    ];

    criticalResources.forEach(resource => {
      const link = document.createElement('link');
      link.rel = 'preload';
      link.as = 'audio';
      link.href = resource;
      document.head.appendChild(link);
    });
  },

  // 初始化所有优化
  init() {
    this.setupLazyLoading();
    this.optimizeAnimations();
    this.setupMemoryManagement();
    this.setupNetworkOptimization();
  }
};

// public/sw.js - Service Worker
const CACHE_NAME = 'fit-app-v1.0.0';
const URLS_TO_CACHE = [
  '/',
  '/index.html',
  '/manifest.json',
  '/static/js/bundle.js',
  '/static/css/main.css'
];

self.addEventListener('install', event => {
  event.waitUntil(
    caches.open(CACHE_NAME)
      .then(cache => {
        console.log('Opened cache');
        return cache.addAll(URLS_TO_CACHE);
      })
  );
});

self.addEventListener('fetch', event => {
  event.respondWith(
    caches.match(event.request)
      .then(response => {
        if (response) {
          return response;
        }
        return fetch(event.request);
      })
  );
});

```bash
#!/bin/bash
# scripts/quick-preview.sh
# Created by Jason Lu on 09:58:00 10/27/2025

set -e

ENVIRONMENT=${1:-development}
PLATFORM=${2:-web}
PORT=${3:-3000}

echo "🚀 Quick Preview for Fit App"
echo "Environment: $ENVIRONMENT"
echo "Platform: $PLATFORM"
echo "Port: $PORT"

# 获取本地IP地址
get_local_ip() {
    if command -v ipconfig >/dev/null 2>&1; then
        ipconfig getifaddr en0 2>/dev/null || echo "localhost"
    elif command -v ip >/dev/null 2>&1; then
        ip route get 1.1.1.1 | awk '{print $7}' || echo "localhost"
    else
        echo "localhost"
    fi
}

# 构建应用
build_app() {
    echo "🔨 Building application..."

    if [ "$ENVIRONMENT" = "development" ]; then
        npm run build:dev
    else
        npm run build
    fi

    echo "✅ Build completed"
}

# 启动预览服务器
start_preview() {
    LOCAL_IP=$(get_local_ip)

    case $PLATFORM in
        "web")
            echo "📱 Starting Web preview server..."
            echo "   Local:   http://localhost:$PORT"
            echo "   Network:  http://$LOCAL_IP:$PORT"
            echo "   QR Code:  https://api.qrserver.com/v1/create-qr-code/?size=200x200&data=http://$LOCAL_IP:$PORT"

            # 生成二维码到终端
            if command -v qrencode >/dev/null 2>&1; then
                echo ""
                qrencode -t ANSI "http://$LOCAL_IP:$PORT"
            fi

            npm run preview
            ;;
        "ios")
            echo "📱 Starting iOS preview..."
            npx cap run ios --livereload --external --host=$LOCAL_IP
            ;;
        "android")
            echo "🤖 Starting Android preview..."
            npx cap run android --livereload --external --host=$LOCAL_IP
            ;;
        "all")
            echo "📱 Starting Web preview for mobile devices..."
            build_app
            npm run preview -- --host 0.0.0.0 --port $PORT
            echo "📱 Access from mobile devices: http://$LOCAL_IP:$PORT"
            ;;
        *)
            echo "❌ Unknown platform: $PLATFORM"
            echo "Usage: $0 [environment] [platform] [port]"
            echo "Platforms: web, ios, android, all"
            exit 1
            ;;
    esac
}

# 清理函数
cleanup() {
    echo "🧹 Cleaning up..."
    pkill -f "npm run preview" || true
    pkill -f "npx cap run" || true
    exit 0
}

# 捕获中断信号
trap cleanup SIGINT SIGTERM

# 主执行流程
build_app
start_preview

# 保持脚本运行
wait
```
